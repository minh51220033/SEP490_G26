function openEditItemModalFromButton(btn) { const itemId = btn.getAttribute('data-item-id'); const staffName = btn.getAttribute('data-staff-name'); const bonus = btn.getAttribute('data-bonus'); const deduction = btn.getAttribute('data-deduction'); const note = btn.getAttribute('data-note'); openEditItemModal(itemId, staffName, bonus, deduction, note); } function openEditItemModal(itemId, staffName, bonus, deduction, note) { document.getElementById('mStaffName').textContent = staffName || ''; document.getElementById('mBonus').value = bonus; document.getElementById('mDeduction').value = deduction; document.getElementById('mNote').value = note || ''; document.getElementById('editItemForm').action = '/payroll/items/' + itemId + '/update'; new bootstrap.Modal(document.getElementById('editItemModal')).show(); }
